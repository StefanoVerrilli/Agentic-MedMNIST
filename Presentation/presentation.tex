  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \small\textbf{The problem}
      \begin{itemize}
        \item Training pipelines are monolithic scripts: hard to audit and to reproduce.
        \item LLM-agent frameworks let a model improvise every step, tied to one vendor.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small\textbf{Our answer}
      \begin{itemize}
        \item One duty per agent; agents exchange only versioned artefacts.
        \item LLM reasoning only at declared decision points, with a fallback.
        \item An independent reviewer audits every stage and can veto.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The task in pictures: PathMNIST}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.82\textheight,keepaspectratio]{fig_samples.png}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small
      \textbf{Can software agents learn to tell colon tissue types apart?}\\[0.4em]
      \begin{itemize}
        \item PathMNIST: stained (H\&E) patches from colorectal cancer slides, part of MedMNIST v2.
        \item 9 tissue classes, 107,180 images of only \textbf{28\,$\times$\,28 pixels}.
        \item Train/validation from one source; test from \textbf{a different hospital}.
        \item Small images keep experiments fast; the second hospital keeps the test honest.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The data at a glance}

    \begin{column}{0.42\textwidth}

      \begin{itemize}
        \item Official split: 89,996 / 10,004 / 7,180 images.
        \item Training classes are near-balanced (largest/smallest = 1.63).
        \item The test hospital differs: adipose 18.6\% vs 10.4\%, debris 4.7\% vs 11.5\%.
        \item So test accuracy measures \textbf{transfer} to new data, not memory.

    \end{column}
  \end{columns}

\begin{frame}{The Blackboard architecture}

    \begin{column}{0.48\textwidth}
      \footnotesize\textbf{Coordination}
      \begin{itemize}
        \item \textbf{Orchestrator}: sequences stages; holds no domain logic.

  \end{columns}

    \end{column}
